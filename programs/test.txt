let rec map f arr =
switch arr =>
  | [] -> []
  | h :: t -> f h :: map f t

let rec filter f arr =
switch arr =>
  | [] -> []
  | h :: t -> if f h then h :: filter f t else filter f t

let rec fold_left f acc lst =
switch lst =>
  | [] -> acc
  | h :: t -> fold_left f (f acc h) t

let rec fold_right f lst acc =
switch lst =>
  | [] -> acc
  | h :: t -> f h (fold_right f t acc)

let numbers = [1 ... 100000]
let strings = "hello" :: "world" :: "lambda" :: "script" :: []

let doubled = map (\x -> x * 2) numbers
let () = println "Doubled numbers:"
let () = println (int_to_str (fold_left (\acc -> \x -> acc + x) 0 doubled))

let even_numbers = filter (\x -> x % 2 == 0) numbers
let () = println "Even numbers:"
let () = println (int_to_str (fold_left (\acc -> \x -> acc + x) 0 even_numbers))

let process_list lst = {
  let doubled = map (\x -> x * 2) lst;
  let sum = fold_left (\acc -> \x -> acc + x) 0 doubled;
  let () = println "Sum of doubled numbers:";
  let () = println (int_to_str sum);
  
  let filtered = filter (\x -> x > 50) doubled;
  let count = fold_left (\acc -> \x -> acc + 1) 0 filtered;
  let () = println "Count of numbers > 50:";
  let () = println (int_to_str count);
  
  count
}

let result = process_list numbers
let () = println "Final result:"
let () = println (int_to_str result)

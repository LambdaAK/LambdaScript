type token_type =
  | TNum of int
  | TPlus
  | TMinus
  | TTimes
  | TDiv

type Option<a> =
  | None
  | Some of a


type expr =
  | EAdd of (expr, expr)
  | ESub of (expr, expr)
  | EMul of (expr, expr)
  | EDiv of (expr, expr)
  | ENum of int


type Parser<a> = [token_type] -> Option<(a, [token_type])>

let (<|>) (p1 : Parser<a>) (p2 : Parser<a>) =
  fn tokens ->
    case p1 tokens do
    | Some result -> Some result
    | None -> p2 tokens

let (<*) (p1 : Parser<a>) (p2 : Parser<b>) =
  fn tokens ->
    case p1 tokens do
    | Some (_, rest) -> p2 rest
    | None -> None

let (*>) p1 p2 =
  fn tokens ->
    case p1 tokens do
    | Some (_, rest) -> p2 rest
    | None -> None

let (>>=) (parser : Parser<a>) (f : a -> Parser<b>) =
  fn tokens ->
    case parser tokens do
    | Some (result, rest) -> f result rest
    | None -> None

let expect_token (expected : token_type) : Parser<unit> =
  fn tokens ->
    case tokens do
    | [] -> None
    | token :: remaining_tokens ->
      if token == expected then Some ((), remaining_tokens) else None

let expect_token_get_data (pred : token_type -> Option<a>) : Parser<a> =
  fn tokens ->
    case tokens do
    | [] -> None
    | token :: remaining_tokens -> (
      case pred token do
      | Some data -> Some (data, remaining_tokens)
      | None -> None
    )


let rec (++) l1 l2 =
  case l1 do
  | [] -> l2
  | h :: t -> h :: (t ++ l2)

let rec rev lst =
  case lst do
  | [] -> []
  | h :: t -> (rev t) ++ [h]

let parse_several (parser : Parser<a>) =
  let rec helper acc tokens =
    case parser tokens do
    | Some (result, remaining_tokens) ->
      helper (result :: acc) remaining_tokens

    | None -> Some (rev acc, tokens)
    in
    helper []


let fail : Parser<a> = fn _ -> None

let return (result : a) : Parser<a> =
  fn tokens ->
    Some (result, tokens)

let num_parser : Parser<expr> =
  fn tokens ->
    case tokens do
    | [] -> None
    | token :: remaining_tokens ->
      case token do
      | TNum n -> Some (ENum n, remaining_tokens)
      | _ -> None

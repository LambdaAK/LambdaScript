type rec Tree<a> =
  | Leaf
  | Node of (a, Tree<a>, Tree<a>)

let rec fold_tree f leaf_case t =
  switch t =>
    | Leaf -> leaf_case
    | Node (x, left, right) ->
        f x (fold_tree f leaf_case left) (fold_tree f leaf_case right)


let rec left_fold_tree f leaf_case t =
          switch t =>
          | Leaf -> leaf_case
          | Node (x, left, right) ->
              let left_acc = left_fold_tree f leaf_case left in
              let curr_acc = f x left_acc in
              left_fold_tree f curr_acc right

type rec List<'a> =
  | Nil
  | Cons of ('a, List<'a>)

type rec Tree<'a> =
  | Leaf
  | Node of ('a, Tree<'a>, Tree<'a>)

let rec fold_tree f acc t =
  case t do
  | Leaf -> acc
  | Node (v, l, r) ->
      let acc_l = fold_tree f acc l in
      let acc_r = fold_tree f acc r in
      f v acc_l acc_r

let sum_tree t = fold_tree (fn a -> fn b -> fn c -> a + b + c) 0 t

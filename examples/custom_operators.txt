/* ============================================
   LambdaScript Custom Binary Operators
   ============================================

   This demonstrates how to define and use
   custom binary operators in LambdaScript.
*/

// Custom operators are defined using parentheses around the operator name
// The operator symbols must start with one of: + - * / % < > =

/* ============================================
   CUSTOM ADDITIVE OPERATORS (precedence like + and -)
   Must start with: + or -
   ============================================ */

// Triple addition operator
let (+++) x y = x + y + y

let result1 = 5 +++ 3  // = 5 + 3 + 3 = 11
let () = println (int_to_str result1)

// Custom concatenation-like operator
let (+-+) a b = a + b + 1

let result2 = 10 +-+ 20  // = 10 + 20 + 1 = 31
let () = println (int_to_str result2)

/* ============================================
   CUSTOM MULTIPLICATIVE OPERATORS (precedence like * / %)
   Must start with: * / or %
   ============================================ */

// Square multiplication operator
let (***) x y = x * x * y

let result3 = 3 *** 2  // = 3 * 3 * 2 = 18
let () = println (int_to_str result3)

// Custom division operator that adds remainder
let (///) x y = x / y + x % y

let result4 = 17 /// 5  // = 17 / 5 + 17 % 5 = 3 + 2 = 5
let () = println (int_to_str result4)

/* ============================================
   CUSTOM RELATIONAL OPERATORS (precedence like < > ==)
   Must start with: < > or =
   ============================================ */

// Custom equality check
let (===) x y = x == y

let result5 = 5 === 5  // = true
let result6 = 5 === 3  // = false
let () = println (if result5 then "true" else "false")
let () = println (if result6 then "true" else "false")

// Custom less-than-or-equal with offset
let (<<) x y = x < y - 1

let result7 = 3 << 5  // = 3 < (5 - 1) = 3 < 4 = true
let () = println (if result7 then "true" else "false")

/* ============================================
   ADVANCED USAGE
   ============================================ */

// Operators can use pattern matching and recursion
let rec (***>) x y =
  if x == 0 then 0
  else if x == 1 then y
  else y + (x - 1) ***> y

let result8 = 4 ***> 3  // = 3 + 3 + 3 + 3 = 12 (like multiplication)
let () = println (int_to_str result8)

// Operators work with polymorphic types too
let (<==>) x y =
  if x == y then 1 else 0

let result9 = 5 <==> 5
let result10 = 3 <==> 7
let () = println (int_to_str result9)  // 1
let () = println (int_to_str result10) // 0

/* ============================================
   OPERATOR RESTRICTIONS
   ============================================ */

// Valid operator starting characters:
// +  -  *  /  %  <  >  =

// Examples of valid custom operators:
// (+++)  (---)  (***)  (///)  (%%%)
// (<<<)  (>>>)  (===)  (=>=)
// (+*+)  (</>) (<+>) (=>>)

// You can mix characters, but the FIRST character
// determines the operator's precedence level!

let (</>) x y = x / y + 1  // Division-level precedence
let (<+>) x y = x + y * 2  // Addition-level precedence

let result11 = 10 </> 3    // = 10 / 3 + 1 = 3 + 1 = 4
let result12 = 10 <+> 3    // = 10 + 3 * 2 = 10 + 6 = 16

let () = println (int_to_str result11)
let () = println (int_to_str result12)

/* End of examples */
